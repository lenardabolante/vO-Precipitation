# ==============================================================================================================
# test_no_windsock_stands_on_a_building_a_street_or_a_pond (removed whole; needs `worlds`, `_spans` and `_overlaps` of the same file)
# From: sim/tests/test_objects.py -- class TestGroundSpots -- lines 287-302 of the file as it was before the windsock was removed.
# ==============================================================================================================
    def test_no_windsock_stands_on_a_building_a_street_or_a_pond(self):
        n_worlds = n_socks = 0
        for label, st, objs in self.worlds():
            n_worlds += 1
            built = _spans(objs, BUILDING_KINDS)
            ponds = _spans(objs, ("pond",))
            streets = [(s["x0"], s["x1"]) for s in st]
            socks = [o for o in objs if o["kind"] == "windsock"]
            self.assertGreaterEqual(len(socks), 1, label)
            for o in socks:
                n_socks += 1
                self.assertFalse(_overlaps(o, built, 8.0), f"windsock on a building {label} x={o['x']}")
                self.assertFalse(_overlaps(o, ponds), f"windsock in a pond {label} x={o['x']}")
                self.assertFalse(_overlaps(o, streets), f"windsock in a street {label} x={o['x']}")
        self.assertGreaterEqual(n_worlds, 60)
        self.assertGreaterEqual(n_socks, 64)
