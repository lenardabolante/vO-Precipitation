// [1] Renderer.java, method drawWindsock (stood after drawSnowpack, before the "clouds" section; lines 1301-1329).
//     Called from drawWorld, see [2]. Uses the Renderer's fields anim (animation clock, seconds) and lit(Color)
//     (daylight), and Model.clamp. Drawn in world coordinates (g has the world transform).
// ==================================================================================================
    private void drawWindsock(Graphics2D g, Model.Obj o) {
        double top = o.y - o.h;
        g.setStroke(new BasicStroke(2.5f, BasicStroke.CAP_ROUND, BasicStroke.JOIN_ROUND));
        g.setColor(lit(new Color(180, 184, 190)));
        g.draw(new Line2D.Double(o.x, o.y, o.x, top));
        double s = Model.clamp(o.sway, -1, 1);
        double droop = (1 - Math.min(1, Math.abs(s) * 1.1)) * 1.25 + Math.sin(anim * 9 + o.x) * 0.06;
        double dir = s >= 0 ? 1 : -1;
        double len = 38;
        double ang = Math.atan2(Math.sin(droop), dir * Math.cos(droop));
        AffineTransform old = g.getTransform();
        g.translate(o.x, top + 3);
        g.rotate(ang);
        int seg = 5;
        for (int k = 0; k < seg; k++) {
            double x0 = len * k / seg, x1 = len * (k + 1) / seg;
            double r0 = 7 - 4.0 * k / seg, r1 = 7 - 4.0 * (k + 1) / seg;
            double wave = Math.sin(anim * 12 - k) * 1.2 * (k / (double) seg);
            Path2D.Double q = new Path2D.Double();
            q.moveTo(x0, -r0 + wave);
            q.lineTo(x1, -r1 + wave);
            q.lineTo(x1, r1 + wave);
            q.lineTo(x0, r0 + wave);
            q.closePath();
            g.setColor(lit(k % 2 == 0 ? new Color(245, 110, 40) : new Color(245, 245, 245)));
            g.fill(q);
        }
        g.setTransform(old);
    }
